import { useState, useEffect, useRef } from "react";
import { invoke } from "@tauri-apps/api/core";
import PlansView from "./components/PlansView";
import FoodsView from "./components/FoodsView";
import ImportView from "./components/ImportView";
import FoodDetailView from "./components/FoodDetailView";
import { isTauri } from "./lib/tauri";
import { useI18n } from "./lib/i18n";
import ThemeToggle from "./components/ThemeToggle";
import PermissionsModal from "./components/PermissionsModal";
import { api } from "./lib/api";
import {
  cancelAllReminders,
  nextWeekdayOccurrence,
  scheduleStepReminder,
} from "./lib/notifications";
import { alarmScheduler, hashId } from "./lib/alarmScheduler";

type View = "plans" | "foods" | "import";

export default function App() {
  const [view, setView] = useState<View>("plans");
  const [openFoodId, setOpenFoodId] = useState<number | null>(null);
  const [navKey, setNavKey] = useState(0);

  const inTauri = isTauri();
  const isAndroid =
    typeof navigator !== "undefined" && /Android/i.test(navigator.userAgent);
  // The alarm-scheduler plugin only registers on Android; gate the UI there.
  const showAlarmUI = inTauri && isAndroid;

  const { t, locale, setLocale } = useI18n();

  // ─── Alarm toggle (Android only; default-on for desktop/web Tauri) ───
  const [alarmsEnabled, setAlarmsEnabled] = useState<boolean>(() => {
    if (!isAndroid) return true;
    try {
      return localStorage.getItem("cookplans.alarmsEnabled") === "1";
    } catch {
      return false;
    }
  });
  const [showPermissions, setShowPermissions] = useState(false);
  const [ringing, setRinging] = useState(false);

  useEffect(() => {
    if (!isAndroid) return;
    try {
      localStorage.setItem(
        "cookplans.alarmsEnabled",
        alarmsEnabled ? "1" : "0"
      );
    } catch { }
  }, [alarmsEnabled, isAndroid]);

  // On launch, if user had enabled alarms, re-verify permissions.
  useEffect(() => {
    if (!showAlarmUI || !alarmsEnabled) return;
    (async () => {
      try {
        const s = await alarmScheduler.checkPermissions();
        const all =
          s.notificationsGranted &&
          s.exactAlarmGranted &&
          s.fullScreenIntentGranted;
        if (!all) setAlarmsEnabled(false);
      } catch {
        /* plugin not installed — ignore */
      }
    })();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [showAlarmUI]);

  // Poll ringing state so we can surface an in-app dismiss banner.
  useEffect(() => {
    if (!showAlarmUI) return;
    let cancelled = false;
    const check = async () => {
      try {
        const s = await invoke<{ ringing: boolean }>(
          "plugin:alarm|is_alarm_ringing"
        );
        if (!cancelled) setRinging(s.ringing);
      } catch { }
    };
    check();
    const id = setInterval(check, 2000);
    return () => {
      cancelled = true;
      clearInterval(id);
    };
  }, [showAlarmUI]);

  const goToView = (v: View) => {
    setOpenFoodId(null);
    setView(v);
    setNavKey((k) => k + 1);
  };

  const tabs: { k: View; label: string; icon: string }[] = [
    { k: "plans", label: t("nav.plans"), icon: "📅" },
    { k: "foods", label: t("nav.foods"), icon: "🥘" },
    { k: "import", label: t("nav.import"), icon: "📥" },
  ];

  // Keep t() fresh inside the long-lived schedule effect.
  const tRef = useRef(t);
  useEffect(() => {
    tRef.current = t;
  });

  // Arm OS-level alarms (Android) / notifications (other Tauri) for the
  // active plan's future cooking steps. Runs on mount, on return from
  // background, and whenever "cookplans:schedules-changed" fires.
  useEffect(() => {
    if (!inTauri) return;
    if (isAndroid && !alarmsEnabled) return;
    let cancelled = false;

    async function reschedule() {
      if (cancelled) return;
      try {
        const plans = await api.listPlans();
        const active = plans.find((p) => p.isActive);

        // Cancel whatever is armed — rebuild from scratch.
        if (isAndroid) await alarmScheduler.cancelAll();
        else await cancelAllReminders();

        if (!active) return;

        const details = await api.getPlanWeekDetails(active.id);
        const now = new Date();

        for (const m of details) {
          const { meal, food, cookStartAt, steps } = m.details;
          for (const step of steps) {
            const fireAt = nextWeekdayOccurrence(
              m.weekday,
              cookStartAt + step.offsetTime
            );
            if (fireAt <= now) continue;

            const key = `${meal.id}|${m.weekday}|${step.id}|${fireAt.getTime()}`;
            const title = tRef.current("notificationContent.stepTitle", {
              stepName: step.name,
            });
            const body = tRef.current("notificationContent.stepBody", {
              foodName: food.name,
              duration: step.duration,
            });
            const largeBody =
              tRef.current("notificationContent.stepLargeBody", {
                stepName: step.name,
                foodName: food.name,
                mealName: meal.name,
                duration: step.duration,
              }) + (step.description ? `\n\n${step.description}` : "");

            if (isAndroid) {
              // Full-screen native alarm: AlarmService (from tauri-plugin-alarm)
              // is started by AlarmManager via PendingIntent.getForegroundService.
              await alarmScheduler.schedule({
                id: hashId(key),
                triggerAtMillis: fireAt.getTime(),
                title,
                body,
                stepName: step.name,
                foodName: food.name,
                dismissLabel: tRef.current("common.dismiss"),
              });
            } else {
              // Desktop / other Tauri: keep using the notification plugin.
              await scheduleStepReminder({ fireAt, key, title, body, largeBody });
            }
          }
        }
      } catch (e) {
        console.warn("[App] reschedule notifications failed", e);
      }
    }

    reschedule();
    const onVis = () => {
      if (!document.hidden) reschedule();
    };
    const onBump = () => reschedule();
    document.addEventListener("visibilitychange", onVis);
    window.addEventListener("cookplans:schedules-changed", onBump);
    return () => {
      cancelled = true;
      document.removeEventListener("visibilitychange", onVis);
      window.removeEventListener("cookplans:schedules-changed", onBump);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [inTauri, isAndroid, alarmsEnabled]);

  async function toggleAlarms(next: boolean) {
    if (next) {
      // Gate on permissions: show the modal, only enable after all granted.
      setShowPermissions(true);
      return;
    }
    // Turning OFF: cancel everything armed and silence any active alarm.
    setAlarmsEnabled(false);
    setRinging(false);
    try {
      if (isAndroid) {
        await alarmScheduler.cancelAll();
        await invoke("plugin:alarm|dismiss");
      } else {
        await cancelAllReminders();
      }
    } catch { }
  }

  return (
    <div className="relative min-h-screen bg-base-100 flex flex-col p-1 py-0">
      {/* ─── Ringing banner: shows while the native alarm is sounding ─── */}
      {ringing && (
        <div className="bg-error text-error-content px-3 py-2 flex items-center gap-2 text-sm border-b-[3px] border-base-content">
          <span className="text-lg">🚨</span>
          <span className="flex-1 font-medium">{t("alarms.ringing")}</span>
          <button
            className="btn btn-xs"
            onClick={async () => {
              try {
                await invoke("plugin:alarm|dismiss");
              } catch { }
              setRinging(false);
            }}
          >
            {t("common.dismiss")}
          </button>
        </div>
      )}

      <header className="sticky top-0 z-30 backdrop-blur bg-base-100/80 border-b border-base-300">
        <div className="max-w-7xl mx-auto px-3 sm:px-4 md:px-6">
          <div className="flex flex-wrap items-center gap-2 sm:gap-3 sm:h-16 py-2 sm:py-0">
            {/* Brand */}
            <h1 className="text-base sm:text-lg md:text-xl font-bold whitespace-nowrap shrink-0">
              🍳 <span className="sm:inline">Cook</span>
              <span className="text-primary">Plans</span>
            </h1>

            {/* Right-side controls — pushed to the end on mobile */}
            <div className="flex items-center gap-1 ms-auto sm:ms-0">
              <ThemeToggle />

              <div
                className="flex items-center gap-1 shrink-0"
                aria-label={t("common.language")}
              >
                <button
                  onClick={() => setLocale("en")}
                  className={`btn btn-xs ${locale === "en" ? "btn-primary" : "btn-ghost"
                    }`}
                  title="English"
                >
                  EN
                </button>
                <button
                  onClick={() => setLocale("fa")}
                  className={`btn btn-xs ${locale === "fa" ? "btn-primary" : "btn-ghost"
                    }`}
                  title="فارسی"
                >
                  فا
                </button>
              </div>

              <span
                className="text-xs opacity-50 hidden md:inline shrink-0 tabular-nums"
                dir="ltr"
              >
                v0.1.0
              </span>
            </div>

            {/* Nav — full row on mobile, inline on ≥sm */}
            <nav className="order-last w-full sm:order-none sm:w-auto sm:flex-1 flex gap-1 overflow-x-auto pb-1 sm:pb-0">
              {tabs.map((tb) => (
                <button
                  key={tb.k}
                  onClick={() => goToView(tb.k)}
                  disabled={!inTauri}
                  className={`btn btn-sm btn-ghost whitespace-nowrap shrink-0 ${view === tb.k && openFoodId === null
                    ? "btn-active bg-base-200"
                    : ""
                    }`}
                >
                  <span>{tb.icon}</span>
                  {tb.label}
                </button>
              ))}
            </nav>
          </div>
        </div>
      </header>

      <main
        className={`flex-1 max-w-7xl w-full mx-auto p-4 md:p-6 pb-13 ${showAlarmUI ? "pb-24" : ""
          }`}
      >
        {!inTauri ? (
          <WebNotice />
        ) : openFoodId !== null ? (
          <FoodDetailView
            foodId={openFoodId}
            onBack={() => setOpenFoodId(null)}
          />
        ) : (
          <>
            {view === "plans" && (
              <PlansView key={`plans-${navKey}`} onOpenFood={setOpenFoodId} />
            )}
            {view === "foods" && (
              <FoodsView key={`foods-${navKey}`} onOpenFood={setOpenFoodId} />
            )}
            {view === "import" && <ImportView />}
          </>
        )}
      </main>

      {/* ─── Footer toggle: Android only (alarm-scheduler plugin) ─── */}
      {showAlarmUI && (
        <footer className="fixed bottom-[var(--safe-area-inset-bottom,0px)] inset-x-0 z-30 backdrop-blur bg-base-100/90 border-t-[3px] h-13 box-border border-base-content">
          <div className="max-w-7xl mx-auto px-3 sm:px-4 py-2 flex items-center justify-between gap-3">
            <label className="flex items-center gap-2 cursor-pointer select-none">
              <span className="text-lg">🔔</span>
              <span className="text-sm font-medium">{t("alarms.toggle")}</span>
              <input
                type="checkbox"
                className="toggle toggle-primary toggle-sm"
                checked={alarmsEnabled}
                onChange={(e) => toggleAlarms(e.target.checked)}
              />
            </label>
            <span
              className={`text-xs tabular-nums ${alarmsEnabled ? "text-success" : "opacity-50"
                }`}
              dir="ltr"
            >
              {alarmsEnabled ? t("alarms.on") : t("alarms.off")}
            </span>
          </div>
        </footer>
      )}

      {/* ─── Permissions gate: shown only when enabling alarms ─── */}
      {showPermissions && (
        <PermissionsModal
          onAllGranted={() => {
            setAlarmsEnabled(true);
            window.dispatchEvent(new Event("cookplans:schedules-changed"));
          }}
          onClose={async () => {
            setShowPermissions(false);
            try {
              const s = await alarmScheduler.checkPermissions();
              if (
                s.notificationsGranted &&
                s.exactAlarmGranted &&
                s.fullScreenIntentGranted
              ) {
                setAlarmsEnabled(true);
                window.dispatchEvent(new Event("cookplans:schedules-changed"));
              }
            } catch (e) {
              console.error("[App] post-modal permission check failed", e);
            }
          }}
        />
      )}
    </div>
  );
}

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useRef,
  useState,
  type ReactNode,
} from "react";

export interface ConfirmOpts {
  message: string;
  title?: string;
  confirmLabel?: string;
  cancelLabel?: string;
  danger?: boolean;
}

type ConfirmFn = (opts: ConfirmOpts) => Promise<boolean>;

const Ctx = createContext<ConfirmFn | null>(null);

interface Pending {
  opts: ConfirmOpts;
  resolve: (v: boolean) => void;
}

export function ConfirmProvider({ children }: { children: ReactNode }) {
  const [pending, setPending] = useState<Pending | null>(null);
  const dialogRef = useRef<HTMLDialogElement>(null);

  const confirm: ConfirmFn = useCallback((opts) => {
    return new Promise<boolean>((resolve) => {
      setPending({ opts, resolve });
    });
  }, []);

  const close = useCallback((v: boolean) => {
    setPending((p) => {
      p?.resolve(v);
      return null;
    });
  }, []);

  useEffect(() => {
    const el = dialogRef.current;
    if (!el) return;
    if (pending) {
      if (!el.open) el.showModal();
    } else {
      if (el.open) el.close();
    }
  }, [pending]);

  return (
    <Ctx.Provider value={confirm}>
      {children}
      <dialog
        ref={dialogRef}
        className="modal"
        onCancel={(e) => {
          e.preventDefault();
          close(false);
        }}
      >
        {pending && (
          <div className="modal-box">
            {pending.opts.title && (
              <h3 className="text-lg font-bold mb-2 break-words">
                {pending.opts.title}
              </h3>
            )}
            <p className="py-2 break-words whitespace-pre-line">
              {pending.opts.message}
            </p>
            <div className="modal-action">
              <button
                type="button"
                className="btn btn-ghost"
                onClick={() => close(false)}
              >
                {pending.opts.cancelLabel ?? "Cancel"}
              </button>
              <button
                type="button"
                className={`btn ${
                  pending.opts.danger ? "btn-error" : "btn-primary"
                }`}
                onClick={() => close(true)}
              >
                {pending.opts.confirmLabel ?? "OK"}
              </button>
            </div>
          </div>
        )}
        <form method="dialog" className="modal-backdrop">
          <button type="button" onClick={() => close(false)}>
            close
          </button>
        </form>
      </dialog>
    </Ctx.Provider>
  );
}

export function useConfirm() {
  const ctx = useContext(Ctx);
  if (!ctx) {
    throw new Error("useConfirm must be used inside ConfirmProvider");
  }
  return ctx;
}

function slugify(s: string) {
  return (
    s.toLowerCase().trim().replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "") || "food"
  );
}

export default function FoodFormModal({
  food,
  onClose,
  onSaved,
}: {
  food: Food | null;
  onClose: () => void;
  onSaved: () => void;
}) {
  const { t } = useI18n();
  const [name, setName] = useState(food?.name ?? "");
  const [desc, setDesc] = useState(food?.description ?? "");
  const [minDur, setMinDur] = useState(food?.minDuration ?? 10);
  const [maxDur, setMaxDur] = useState(food?.maxDuration ?? 20);
  const [err, setErr] = useState<string | null>(null);

  const isNew = !food;

  async function save() {
    if (!name.trim()) {
      setErr(t("common.nameRequired"));
      return;
    }

      if (isNew) {
        await api.createFood({
          name: name.trim(),
          slug: slugify(name),
          description: desc.trim() || undefined,
          minDuration: Number(minDur) || 0,
          maxDuration: Number(maxDur) || 0,

          id: food!.id,
          name: name.trim(),
          slug: food!.slug,
          description: desc.trim() || undefined,
          minDuration: Number(minDur) || 0,
          maxDuration: Number(maxDur) || 0,
        });

    } catch (e) {
      setErr(String(e));

async function copyText(text: string): Promise<boolean> {
  try {
    if (navigator.clipboard && window.isSecureContext) {
      await navigator.clipboard.writeText(text);
      return true;
    }
  } catch {}
  try {
    const ta = document.createElement("textarea");
    ta.value = text;
    ta.style.position = "fixed";
    ta.style.opacity = "0";
    document.body.appendChild(ta);
    ta.select();
    const ok = document.execCommand("copy");
    document.body.removeChild(ta);
    return ok;
  } catch {
    return false;
  }
}

import { useEffect, useState } from "react";
import { invoke } from "@tauri-apps/api/core";
import { useI18n } from "../lib/i18n";
import { alarmScheduler, type PermissionsStatus } from "../lib/alarmScheduler";

export default function PermissionsModal({
  onClose,
  onAllGranted,
}: {
  onClose: () => void;
  onAllGranted?: () => void;
}) {
  const { t } = useI18n();
  const [status, setStatus] = useState<PermissionsStatus | null>(null);
  const [busy, setBusy] = useState<string | null>(null);

  async function refresh() {
    try {
      const s = await alarmScheduler.checkPermissions();
      setStatus(s);
      if (s.notificationsGranted && s.exactAlarmGranted && s.fullScreenIntentGranted) {
        onAllGranted?.();
      }
    } catch (e) {
      console.error("[PermissionsModal] checkPermissions failed", e);
      setStatus({
        notificationsGranted: false,
        exactAlarmGranted: false,
        fullScreenIntentGranted: false,
      });
      // Do NOT call onAllGranted() — we don't actually know the state.

      // Non-Android target: treat as granted so the toggle can enable.
      // setStatus({ notificationsGranted: true, exactAlarmGranted: true, fullScreenIntentGranted: true });
      // onAllGranted?.();
    }
  }
  useEffect(() => { refresh(); }, []);

  const [err, setErr] = useState<string | null>(null);

  async function grant(kind: "notif" | "exact" | "fsi") {
    setBusy(kind);
    setErr(null);
    try {
      if (kind === "notif") {
        await alarmScheduler.requestNotificationPermission();
      } else if (kind === "exact") {
        await invoke("plugin:alarm|open_exact_alarm_settings");
      } else {
        await invoke("plugin:alarm|open_full_screen_intent_settings");
      }
      setTimeout(refresh, 800);
    } catch (e) {
      console.error(`[PermissionsModal] grant(${kind}) failed`, e);
      setErr(String(e));
    } finally {
      setBusy(null);
    }
  }

  return (
    <div className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4" onClick={onClose}>
      <div
        className="bg-base-100 border-[3px] border-base-content shadow-[6px_6px_0_var(--color-base-content)] w-full max-w-md"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="p-6 space-y-4">
          <h2 className="text-xl font-bold flex items-center gap-2">
            <span className="text-3xl">🔔</span> {t("permissions.title")}
          </h2>
          <p className="text-sm opacity-80">{t("permissions.body")}</p>
          {err && (
            <div className="alert alert-error text-xs break-words" dir="ltr">
              {err}
            </div>
          )}
          <Row label={t("permissions.notifications")} ok={status?.notificationsGranted ?? false}
            busy={busy === "notif"} grantLabel={t("permissions.grant")}
            grantedLabel={t("permissions.granted")} onGrant={() => grant("notif")} />
          <Row label={t("permissions.exactAlarm")} ok={status?.exactAlarmGranted ?? false}
            busy={busy === "exact"} grantLabel={t("permissions.openSettings")}
            grantedLabel={t("permissions.granted")} onGrant={() => grant("exact")} />
          <Row label={t("permissions.fullScreenIntent")} ok={status?.fullScreenIntentGranted ?? false}
            busy={busy === "fsi"} grantLabel={t("permissions.openSettings")}
            grantedLabel={t("permissions.granted")} onGrant={() => grant("fsi")} />

          <div className="flex justify-end">
            <button className="btn btn-ghost btn-sm" onClick={onClose}>{t("common.close")}</button>
          </div>
        </div>
      </div>
    </div>
  );
}

function Row({
  label, ok, busy, grantLabel, grantedLabel, onGrant,
}: { label: string; ok: boolean; busy: boolean; grantLabel: string; grantedLabel: string; onGrant: () => void; }) {
  return (
    <div className={`flex items-center gap-3 p-3 border-2 ${ok ? "border-success bg-success/10" : "border-base-300"}`}>
      <div className="flex-1 text-sm font-medium">{label}</div>
      {ok ? (
        <span className="badge badge-success badge-sm">✓ {grantedLabel}</span>
      ) : (
        <button className="btn btn-xs btn-primary" onClick={onGrant} disabled={busy}>
          {busy ? <span className="loading loading-spinner loading-xs" /> : grantLabel}
        </button>
      )}
    </div>
  );
}

import { useEffect, useMemo, useState } from "react";
import { api } from "../lib/api";
import type { Plan, Food, PlanMealDetail } from "../lib/types";
import { MEAL_EMOJI, timeToMinutes, minutesToInput } from "../lib/types";
import { useI18n } from "../lib/i18n";
import { useConfirm } from "./ConfirmDialog";

export default function WeekView({
  plan,
  onPlanUpdated,
  onOpenFood,
}: {
  plan: Plan;
  onPlanUpdated: (p: Plan) => void;
  onOpenFood: (id: number) => void;
}) {
  const { t, weekdayName, formatTime } = useI18n();
  const askConfirm = useConfirm();
  const [weekDetails, setWeekDetails] = useState<PlanMealDetail[]>([]);
  const [foods, setFoods] = useState<Food[]>([]);
  const [focusedDay, setFocusedDay] = useState<number>(() => new Date().getDay());
  const [editing, setEditing] = useState<{ weekday: number; detail?: PlanMealDetail } | null>(null);
  const [err, setErr] = useState<string | null>(null);
  const [nowMin, setNowMin] = useState(() => {
    const d = new Date();
    return d.getHours() * 60 + d.getMinutes() + d.getSeconds() / 60;
  });

  const dayOrder = useMemo(() => {
    const arr: number[] = [];
    for (let i = 0; i < 7; i++) arr.push((plan.startWeekday + i) % 7);
    return arr;
  }, [plan.startWeekday]);

  async function refresh() {
    try {
      const [details, fs] = await Promise.all([
        api.getPlanWeekDetails(plan.id),
        api.listFoods(),
      ]);
      setWeekDetails(details);
      setFoods(fs);
    } catch (e) {
      setErr(String(e));
    }
  }

  useEffect(() => {
    refresh();
  }, [plan.id]);

  useEffect(() => {
    const t = setInterval(() => {
      const d = new Date();
      setNowMin(d.getHours() * 60 + d.getMinutes() + d.getSeconds() / 60);
    }, 1000);
    return () => clearInterval(t);
  }, []);

  const mealCountByDay = useMemo(() => {
    const m: Record<number, number> = {};
    for (const d of weekDetails) m[d.weekday] = (m[d.weekday] ?? 0) + 1;
    return m;
  }, [weekDetails]);

  const focusedMeals = weekDetails
    .filter((d) => d.weekday === focusedDay)
    .sort((a, b) => a.details.meal.deadlineAt - b.details.meal.deadlineAt);

  const allSteps = focusedMeals.flatMap((m) =>
    m.details.steps.map((s) => ({
      ...s,
      mealName: m.details.meal.name,
      foodName: m.details.food.name,
      planMealId: m.planMealId,
    }))
  );
  const currentStep = allSteps
    .filter((s) => s.startAt <= nowMin && s.endAt > nowMin)
    .sort((a, b) => a.startAt - b.startAt)[0];
  const nextStep = allSteps
    .filter((s) => s.startAt > nowMin)
    .sort((a, b) => a.startAt - b.startAt)[0];

  return (
    <div className="space-y-4">
      {err && <div className="alert alert-error text-sm">{err}</div>}

      <div className="flex flex-wrap items-center gap-2 text-sm">
        <span className="opacity-60">{t("week.weekStartsOn")}</span>
        <select
          className="select select-sm select-bordered"
          value={plan.startWeekday}
          onChange={async (e) => {
            const val = Number(e.target.value);
            await api.updatePlan({
              id: plan.id,
              name: plan.name,
              description: plan.description ?? undefined,
              slug: plan.slug,
              startWeekday: val,
            });
            onPlanUpdated({ ...plan, startWeekday: val });
            window.dispatchEvent(new Event("cookplans:schedules-changed"));
          }}
        >
          {[6, 0, 1, 2, 3, 4, 5].map((i) => (
            <option key={i} value={i}>
              {weekdayName(i)}
            </option>
          ))}
        </select>
      </div>

      <div className="grid grid-cols-2 sm:grid-cols-4 lg:grid-cols-7 gap-2">
        {dayOrder.map((weekday) => {
          const count = mealCountByDay[weekday] ?? 0;
          const isFocused = focusedDay === weekday;
          const isToday = new Date().getDay() === weekday;
          return (
            <button
              key={weekday}
              onClick={() => setFocusedDay(weekday)}
              className={`card border-2 p-2 sm:p-3 flex flex-col items-center justify-center min-h-[92px] transition ${isFocused
                  ? "bg-primary/20 border-primary shadow-lg"
                  : "bg-base-200 border-base-300 hover:border-primary/50"
                }`}
            >
              <div
                className={`text-[10px] sm:text-xs uppercase tracking-wider sm:tracking-widest font-semibold text-center break-words leading-tight ${isFocused ? "text-primary" : "opacity-60"
                  }`}
              >
                {weekdayName(weekday)}
                {isToday && <span className="ms-1 text-[10px] text-success">●</span>}
              </div>
              <div
                className={`text-3xl font-bold leading-none mt-1 tabular-nums ${count === 0 ? "opacity-30" : ""
                  }`}
                dir="ltr"
              >
                {count}
              </div>
              <div className="text-[10px] opacity-50 mt-0.5 whitespace-nowrap">
                {count === 1 ? t("week.mealSingular") : t("week.mealPlural")}
              </div>
            </button>
          );
        })}
      </div>

      <div className="space-y-3 mt-4">
        <div className="flex items-center gap-2 flex-wrap">
          <h3 className="font-bold text-lg break-words min-w-0">
            {t("week.dayMeals", { day: weekdayName(focusedDay) })}
          </h3>
          <span className="badge badge-ghost tabular-nums shrink-0" dir="ltr">
            {focusedMeals.length}
          </span>
          <div className="flex-1" />
          <button
            className="btn btn-primary btn-sm shrink-0"
            onClick={() => setEditing({ weekday: focusedDay })}
          >
            {t("week.addMeal")}
          </button>
        </div>

        {currentStep && (
          <div className="card bg-success/10 border-2 border-success/40">
            <div className="card-body p-4 flex-row items-center gap-4 flex-wrap">
              <div className="text-3xl shrink-0">🍳</div>
              <div className="flex-1 min-w-[180px]">
                <div className="text-xs uppercase tracking-wider text-success font-bold">
                  {t("week.currentMeal")}
                </div>
                <div className="font-semibold break-words text-base sm:text-lg">
                  {currentStep.foodName} · {currentStep.name}
                </div>
                <div className="rtl:text-right mt-1 flex flex-wrap gap-2 items-baseline">
                  <span dir="ltr">
                    <bdi className="text-lg font-semibold tabular-nums">
                      {formatTime(currentStep.startAt)}
                    </bdi>
                    <span className="opacity-50">→</span>
                    <bdi className="text-lg font-semibold tabular-nums opacity-80">
                      {formatTime(currentStep.endAt)}
                    </bdi>
                    <bdi className="text-sm opacity-60 tabular-nums">
                      ({currentStep.duration}m)
                    </bdi>
                  </span>
                </div>
              </div>
              <div className="text-end shrink-0">
                <div className="text-xs uppercase tracking-wider opacity-60">
                  {t("week.mealLabel")}
                </div>
                <div className="font-semibold text-sm break-words">
                  {currentStep.mealName}
                </div>
              </div>
            </div>
          </div>
        )}

        {nextStep && (
          <div className="card bg-gradient-to-br from-primary/20 to-secondary/10 border border-primary/30">
            <div className="card-body p-4 flex-row items-center gap-4 flex-wrap">
              <div className="text-3xl shrink-0">⏳</div>
              <div className="flex-1 min-w-[180px]">
                <div className="text-xs uppercase tracking-wider opacity-70">
                  {t("week.nextStep")}
                </div>
                <div className="font-semibold break-words text-base sm:text-lg">
                  {nextStep.foodName} · {nextStep.name}
                </div>
                <div className="mt-1 flex flex-wrap gap-2 items-baseline">
                  <span className="text-xs uppercase tracking-wider opacity-60">
                    {t("week.at")}
                  </span>
                  <bdi dir="ltr" className="text-2xl font-bold tabular-nums text-primary">
                    {formatTime(nextStep.startAt)}
                  </bdi>
                  <bdi dir="ltr" className="text-sm opacity-70 tabular-nums">
                    ({nextStep.duration}m)
                  </bdi>
                </div>
              </div>
              <div className="text-end shrink-0">
                <div className="text-xs uppercase tracking-wider opacity-70">{t("week.in")}</div>
                <div className="text-3xl font-mono font-bold tabular-nums" dir="ltr">
                  {(() => {
                    const total = Math.max(0, Math.floor((nextStep.startAt - nowMin) * 60));
                    const mm = Math.floor(total / 60);
                    const ss = total % 60;
                    return `${String(mm).padStart(2, "0")}:${String(ss).padStart(2, "0")}`;
                  })()}
                </div>
              </div>
            </div>
          </div>
        )}

        <div className="flex flex-col gap-3">
          {focusedMeals.length === 0 && (
            <div className="card bg-base-200 border-2 border-dashed border-base-300 p-8 text-center">
              <div className="opacity-60 mb-3 break-words">
                {t("week.noMeals", { day: weekdayName(focusedDay) })}
              </div>
              <button
                className="btn btn-primary btn-sm"
                onClick={() => setEditing({ weekday: focusedDay })}
              >
                {t("week.addMeal")}
              </button>
            </div>
          )}
          {focusedMeals.map((m) => (
            <MealRow
              key={m.planMealId}
              detail={m}
              nowMin={nowMin}
              onOpenFood={onOpenFood}
              onEdit={() => setEditing({ weekday: focusedDay, detail: m })}
              onDelete={async () => {
                const ok = await askConfirm({
                  message: t("week.deleteMealConfirm", {
                    name: m.details.meal.name,
                  }),
                  danger: true,
                  confirmLabel: t("common.delete"),
                  cancelLabel: t("common.cancel"),
                });
                if (!ok) return;
                await api.deleteMeal(m.details.meal.id);
                refresh();
                window.dispatchEvent(new Event("cookplans:schedules-changed"));
              }}
            />
          ))}
        </div>
      </div>

      {editing && (
        <MealEditor
          plan={plan}
          weekday={editing.weekday}
          existing={editing.detail}
          foods={foods}
          onClose={() => setEditing(null)}
          onSaved={() => {
            setEditing(null);
            refresh();
            window.dispatchEvent(new Event("cookplans:schedules-changed"));
          }}
        />
      )}
    </div>
  );
}

function MealRow({

  nowMin,
  onOpenFood,
  onEdit,
  onDelete,

  detail: PlanMealDetail;
  nowMin: number;
  onOpenFood: (id: number) => void;
  onEdit: () => void;

  const currentStep = steps.find((s) => s.startAt <= nowMin && s.endAt > nowMin);

  return (
    <div className="card bg-base-200 border border-base-300">
      <div className="card-body p-4 min-w-0">
        <div className="flex items-center gap-2 flex-wrap">
          <span className="text-2xl shrink-0">{MEAL_EMOJI[meal.mealType] ?? "🍽"}</span>

                      ? "bg-primary/20 border border-primary/50"
                      : isPast
                        ? "opacity-40"
                        : "bg-base-300/40"

                  </bdi>

              );

          </div>

function MealEditor({
  plan,
  weekday,
  existing,
  foods,
  onClose,
  onSaved,
}: {
  plan: Plan;
  weekday: number;
  existing?: PlanMealDetail;
  foods: Food[];
  onClose: () => void;
  onSaved: () => void;
}) {
  const { t, weekdayName } = useI18n();
  const [name, setName] = useState(existing?.details.meal.name ?? "");
  const [desc, setDesc] = useState(existing?.details.meal.description ?? "");
  const [type, setType] = useState(existing?.details.meal.mealType ?? "breakfast");
  const [deadline, setDeadline] = useState(
    existing ? minutesToInput(existing.details.meal.deadlineAt) : "08:00"
  );
  const [foodId, setFoodId] = useState<number>(
    existing?.details.meal.foodId ?? (foods[0]?.id ?? 0)
  );
  const [err, setErr] = useState<string | null>(null);
  const [busy, setBusy] = useState(false);

  async function save() {
    if (!name.trim()) {
      setErr(t("common.nameRequired"));
      return;
    }
    if (!foodId) {
      setErr(t("week.pickFood"));
      return;
    }
    setBusy(true);
    setErr(null);
    try {
      if (existing) {
        await api.updateMeal({
          mealId: existing.details.meal.id,
          name: name.trim(),
          description: desc.trim() || undefined,
          mealType: type,
          deadlineAt: timeToMinutes(deadline),
          foodId,
        });
      } else {
        await api.createMeal({
          planId: plan.id,
          weekday,
          name: name.trim(),
          description: desc.trim() || undefined,
          mealType: type,
          deadlineAt: timeToMinutes(deadline),
          foodId,
        });
      }
      onSaved();
    } catch (e) {
      setErr(String(e));
    } finally {
      setBusy(false);
    }
  }

  return (
    <div
      className="fixed inset-0 z-50 bg-black/60 flex items-center justify-center p-4"
      onClick={onClose}
    >
      <div
        className="bg-base-100 border-[3px] border-base-content shadow-[6px_6px_0_var(--color-base-content)] w-full max-w-lg max-h-[90vh] overflow-y-auto overflow-x-hidden"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="p-6 space-y-4">
          <div className="flex items-center gap-2 flex-wrap">
            <span className="text-3xl shrink-0">{MEAL_EMOJI[type] ?? "🍽"}</span>
            <h2 className="text-xl font-bold break-words min-w-0">
              {existing ? t("week.editMeal") : t("week.newMeal")}{" "}
              <span className="opacity-50 font-normal">· {weekdayName(weekday)}</span>
            </h2>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
            <div className="form-control md:col-span-2">
              <label className="label">
                <span className="label-text">{t("common.name")}</span>
              </label>
              <input
                className="input input-bordered w-full"
                value={name}
                onChange={(e) => setName(e.target.value)}
                autoFocus
              />
            </div>
            <div className="form-control">
              <label className="label">
                <span className="label-text">{t("week.type")}</span>
              </label>
              <select
                className="select select-bordered w-full"
                value={type}
                onChange={(e) => setType(e.target.value)}
              >
                <option value="breakfast">🌅 {t("week.mealTypes.breakfast")}</option>
                <option value="lunch">☀️ {t("week.mealTypes.lunch")}</option>
                <option value="dinner">🌙 {t("week.mealTypes.dinner")}</option>
                <option value="snack">🍪 {t("week.mealTypes.snack")}</option>
              </select>
            </div>
            <div className="form-control">
              <label className="label">
                <span className="label-text">{t("week.readyTime")}</span>
              </label>
              <input
                type="time"
                className="input input-bordered w-full"
                dir="ltr"
                value={deadline}
                onChange={(e) => setDeadline(e.target.value)}
              />
            </div>
            <div className="form-control md:col-span-2">
              <label className="label">
                <span className="label-text">{t("week.food")}</span>
              </label>
              {foods.length === 0 ? (
                <div className="text-sm opacity-60">{t("week.noFoodsYet")}</div>
              ) : (
                <select
                  className="select select-bordered w-full"
                  value={foodId}
                  onChange={(e) => setFoodId(Number(e.target.value))}
                >
                  {foods.map((f) => (
                    <option key={f.id} value={f.id}>
                      {f.name} ({f.maxDuration}m)
                    </option>
                  ))}
                </select>
              )}
            </div>
            <div className="form-control md:col-span-2">
              <label className="label">
                <span className="label-text">{t("common.description")}</span>
              </label>
              <textarea
                className="textarea textarea-bordered w-full"
                rows={2}
                value={desc}
                onChange={(e) => setDesc(e.target.value)}
              />
            </div>
          </div>

          {err && <div className="alert alert-error text-sm">{err}</div>}

          <div className="flex justify-end gap-2">
            <button className="btn btn-ghost btn-sm" onClick={onClose}>
              {t("common.cancel")}
            </button>
            <button className="btn btn-primary btn-sm" onClick={save} disabled={busy}>
              {busy ? <span className="loading loading-spinner loading-xs" /> : t("common.save")}
            </button>
          </div>
        </div>
      </div>
    </div>
  );
}

===== lib/i18n.tsx =====
import React, { createContext, useContext, useEffect, useState } from "react";
import en from "../locales/en.json";
import fa from "../locales/fa.json";

export type Locale = "en" | "fa";

const dicts: Record<Locale, any> = { en, fa };

interface I18nCtx {
  locale: Locale;
  setLocale: (l: Locale) => void;
  t: (key: string, vars?: Record<string, string | number>) => string;
  weekdayName: (day: number) => string;
  weekdayShortName: (day: number) => string;
  formatTime: (minutes: number | null | undefined) => string;
}

const Ctx = createContext<I18nCtx | null>(null);

const WEEKDAY_KEYS = ["sun", "mon", "tue", "wed", "thu", "fri", "sat"];
const WEEKDAY_SHORT_KEYS = [
  "sunShort",
  "monShort",
  "tueShort",
  "wedShort",
  "thuShort",
  "friShort",
  "satShort",
];

function get(obj: any, path: string): any {
  return path.split(".").reduce((a, k) => (a == null ? undefined : a[k]), obj);
}

function interpolate(s: string, vars?: Record<string, string | number>) {
  if (!vars) return s;
  return s.replace(/\{\{(\w+)\}\}/g, (_, k) => String(vars[k] ?? ""));
}

export function I18nProvider({ children }: { children: React.ReactNode }) {
  const [locale, setLocaleState] = useState<Locale>(() => {
    try {
      const saved = localStorage.getItem("cookplans.locale");
      return saved === "fa" ? "fa" : "en";
    } catch {
      return "en";
    }
  });

  const setLocale = (l: Locale) => {
    setLocaleState(l);
    try {
      localStorage.setItem("cookplans.locale", l);
    } catch {}
  };

  useEffect(() => {
    document.documentElement.lang = locale;
    document.documentElement.dir = locale === "fa" ? "rtl" : "ltr";
  }, [locale]);

  const t = (key: string, vars?: Record<string, string | number>) => {
    const d = dicts[locale];
    const val = get(d, key) ?? get(dicts.en, key) ?? key;
    return interpolate(String(val), vars);
  };

  const weekdayName = (day: number) =>
    t(`weekdays.${WEEKDAY_KEYS[((day % 7) + 7) % 7]}`);
  const weekdayShortName = (day: number) =>
    t(`weekdays.${WEEKDAY_SHORT_KEYS[((day % 7) + 7) % 7]}`);

  const formatTime = (minutes: number | null | undefined) => {
    if (minutes == null) return "—";
    const mm = ((Math.round(minutes) % 1440) + 1440) % 1440;
    const h = Math.floor(mm / 60);
    const m = mm % 60;
    if (locale === "fa") {
      return `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}`;
    }
    const ampm = h < 12 ? "AM" : "PM";
    const h12 = h === 0 ? 12 : h > 12 ? h - 12 : h;
    return `${h12}:${String(m).padStart(2, "0")} ${ampm}`;
  };

  return (
    <Ctx.Provider value={{ locale, setLocale, t, weekdayName, weekdayShortName, formatTime }}>
      {children}
    </Ctx.Provider>
  );
}

export function useI18n() {
  const ctx = useContext(Ctx);
  if (!ctx) throw new Error("useI18n must be used inside I18nProvider");
  return ctx;
}

import {

  createChannel,
  Importance,
  pending,
  Schedule,
  sendNotification,
  Visibility,          // ← added

const CHANNEL_ID = "cookplans-reminders";
const CHANNEL_NAME = "Cooking reminders";
const CHANNEL_DESCRIPTION = "Step-by-step cooking alerts";

let channelReady = false;

/**
 * Ensure the high-importance Android notification channel exists.
 * Importance.High gives us the system default sound + heads-up banner.
 * Visibility.Public shows content on the lock screen.
 * Safe to call multiple times; it is a no-op after the first success.
 */
export async function ensureCookingChannel(): Promise<void> {
  if (channelReady) return;
  try {
    await createChannel({
      id: CHANNEL_ID,
      name: CHANNEL_NAME,
      description: CHANNEL_DESCRIPTION,
      importance: Importance.High,
      visibility: Visibility.Public,   // ← fixed: Visibility, not Importance
      vibration: true,
    });
    channelReady = true;
  } catch (e) {
    console.warn("[notifications] createChannel failed", e);
  }
}

function hashId(s: string): number {
  let h = 2166136261;
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return Math.abs(h | 0);
}

export interface StepReminderParams {
  fireAt: Date;
  key: string;
  title: string;
  body: string;
  largeBody: string;
}

/**
 * Schedule an OS-level notification via Android's AlarmManager.
 * Fires even if the app is closed or the screen is locked.
 * allowWhileIdle so Doze mode doesn't defer it by hours.
 */
export async function scheduleStepReminder(p: StepReminderParams): Promise<void> {
  await ensureCookingChannel();
  const id = hashId(p.key);
  try {
    await sendNotification({
      id,
      title: p.title,
      body: p.body,
      largeBody: p.largeBody,
      channelId: CHANNEL_ID,
      schedule: Schedule.at(p.fireAt, false, true),
    });
  } catch (e) {
    console.warn("[notifications] schedule failed", e);
  }
}

/**
 * Remove every pending (not-yet-fired) scheduled notification.
 * Called before rescheduling so the OS queue matches the DB.
 */
export async function cancelAllReminders(): Promise<void> {
  try {
    const armed = await pending();
    if (armed.length) {
      // ← fixed: cancel() expects number[] — map to IDs
      await cancel(armed.map((n) => n.id));
    }
  } catch (e) {
    console.warn("[notifications] cancelAll failed", e);
  }
}

/**
 * Return the next Date at the given weekday (0=Sun..6=Sat) and time
 * (minutes from midnight). Skips to next week if it has already passed.
 */
export function nextWeekdayOccurrence(
  targetDow: number,
  minutesFromMidnight: number
): Date {
  const now = new Date();
  const target = new Date(now);
  target.setHours(0, 0, 0, 0);
  const diff = (targetDow - now.getDay() + 7) % 7;
  target.setDate(now.getDate() + diff);
  target.setHours(
    Math.floor(minutesFromMidnight / 60),
    minutesFromMidnight % 60,
    0,
    0
  );
  if (target <= now) target.setDate(target.getDate() + 7);
  return target;
}

===== lib/useTheme.ts =====
import { useEffect, useState } from "react";

export type Theme = "neobrutalism" | "neobrutalism-dark";

const STORAGE_KEY = "cookplans.theme";

export function useTheme() {
  const [theme, setTheme] = useState<Theme>(() => {
    try {
      const stored = localStorage.getItem(STORAGE_KEY);
      if (stored === "neobrutalism" || stored === "neobrutalism-dark") {
        return stored;
      }
    } catch {}
    return window.matchMedia("(prefers-color-scheme: dark)").matches
      ? "neobrutalism-dark"
      : "neobrutalism";
  });

  useEffect(() => {
    document.documentElement.setAttribute("data-theme", theme);
    try {
      localStorage.setItem(STORAGE_KEY, theme);
    } catch {}
  }, [theme]);

  const toggleTheme = () => {
    setTheme((t) =>
      t === "neobrutalism" ? "neobrutalism-dark" : "neobrutalism"
    );
  };

  return { theme, toggleTheme, setTheme };
}
